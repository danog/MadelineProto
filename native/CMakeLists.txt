cmake_minimum_required(VERSION 3.16)
project(madeline_crypto LANGUAGES C)
find_package(OpenSSL 1.1 REQUIRED COMPONENTS Crypto)
add_library(madeline-crypto SHARED ige.c)
# Optional asynchronous executor; the synchronous helper also builds on Windows.
if(UNIX)
    find_package(Threads REQUIRED)
    target_sources(madeline-crypto PRIVATE crypto.c)
    target_link_libraries(madeline-crypto PRIVATE Threads::Threads m)
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" CRYPTO_PROCESSOR)
    if(CRYPTO_PROCESSOR MATCHES "^(aarch64|arm64)$")
        target_sources(madeline-crypto PRIVATE aes_arm.c)
        set_source_files_properties(aes_arm.c PROPERTIES COMPILE_OPTIONS "-march=armv8-a+crypto")
        target_compile_definitions(madeline-crypto PRIVATE MADELINE_ARM_AES)
    elseif(CRYPTO_PROCESSOR MATCHES "^(x86_64|amd64)$")
        target_sources(madeline-crypto PRIVATE aes_x86.c)
        set_source_files_properties(aes_x86.c PROPERTIES COMPILE_OPTIONS "-maes;-msse2")
        target_compile_definitions(madeline-crypto PRIVATE MADELINE_X86_AES)
    endif()
endif()
target_compile_features(madeline-crypto PRIVATE c_std_99)
target_link_libraries(madeline-crypto PRIVATE OpenSSL::Crypto)
set_target_properties(madeline-crypto PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)

include(CTest)
if(BUILD_TESTING AND UNIX AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/native/policy.c")
    add_executable(crypto-policy-test ../tests/native/policy.c)
    target_include_directories(crypto-policy-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(crypto-policy-test PRIVATE m)
    add_test(NAME crypto-policy COMMAND crypto-policy-test)
    add_executable(crypto-executor-test ../tests/native/executor.c)
    target_include_directories(crypto-executor-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(crypto-executor-test PRIVATE madeline-crypto)
    add_test(NAME crypto-executor COMMAND crypto-executor-test)
    if(CRYPTO_PROCESSOR MATCHES "^(aarch64|arm64|x86_64|amd64)$")
        add_executable(crypto-ige-test ../tests/native/ige.c)
        target_link_libraries(crypto-ige-test PRIVATE madeline-crypto)
        if(CRYPTO_PROCESSOR MATCHES "^(aarch64|arm64)$")
            target_compile_definitions(crypto-ige-test PRIVATE MADELINE_ARM_AES)
        else()
            target_compile_definitions(crypto-ige-test PRIVATE MADELINE_X86_AES)
        endif()
        add_test(NAME crypto-ige COMMAND crypto-ige-test)
        set_tests_properties(crypto-ige PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endif()

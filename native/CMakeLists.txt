cmake_minimum_required(VERSION 3.16)
project(madeline_ige LANGUAGES C)
find_package(OpenSSL 1.1 REQUIRED COMPONENTS Crypto)
add_library(madeline-ige SHARED ige.c)
# Optional asynchronous executor; the synchronous helper also builds on Windows.
if(UNIX)
    find_package(Threads REQUIRED)
    target_sources(madeline-ige PRIVATE crypto.c)
    target_link_libraries(madeline-ige PRIVATE Threads::Threads m)
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" CRYPTO_PROCESSOR)
    if(CRYPTO_PROCESSOR MATCHES "^(aarch64|arm64)$")
        target_sources(madeline-ige PRIVATE aes_arm.c)
        set_source_files_properties(aes_arm.c PROPERTIES COMPILE_OPTIONS "-march=armv8-a+crypto")
        target_compile_definitions(madeline-ige PRIVATE MADELINE_ARM_AES)
    elseif(CRYPTO_PROCESSOR MATCHES "^(x86_64|amd64)$")
        target_sources(madeline-ige PRIVATE aes_x86.c)
        set_source_files_properties(aes_x86.c PROPERTIES COMPILE_OPTIONS "-maes;-msse2")
        target_compile_definitions(madeline-ige PRIVATE MADELINE_X86_AES)
    endif()
endif()
target_compile_features(madeline-ige PRIVATE c_std_99)
target_link_libraries(madeline-ige PRIVATE OpenSSL::Crypto)
set_target_properties(madeline-ige PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
